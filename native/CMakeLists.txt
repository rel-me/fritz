cmake_minimum_required(VERSION 3.20)
project(fritz_llama LANGUAGES C CXX)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
foreach(option LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER LLAMA_BUILD_APP LLAMA_BUILD_UI LLAMA_OPENSSL LLAMA_SUBPROCESS GGML_BLAS GGML_OPENMP GGML_NATIVE GGML_CPU_KLEIDIAI GGML_CCACHE)
    set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(GGML_METAL ON CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
add_subdirectory(../vendor/llama.cpp llama)
add_library(fritz_llama STATIC bridge.cpp)
target_compile_features(fritz_llama PRIVATE cxx_std_17)
target_link_libraries(fritz_llama PRIVATE llama-common)
